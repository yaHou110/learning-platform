## 2. Drizzle ORM + PostgreSQL Patterns

### Connection & Pooling (Payload CMS Pattern — Production Ready)

```typescript
// packages/core/src/db/connect.ts
import { drizzle } from 'drizzle-orm/node-postgres'
import { Pool } from 'pg'
import * as schema from './schema'

const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  max: 20,                    // Pool size
  idleTimeoutMillis: 30000,   // Close idle connections
  connectionTimeoutMillis: 2000,
})

// Read replicas support (Payload pattern)
const readReplicas = process.env.DATABASE_READ_REPLICAS?.split(',') || []
const readPools = readReplicas.map(url => new Pool({ connectionString: url, max: 10 }))

export const db = drizzle({
  client: pool,
  schema,
  logger: process.env.NODE_ENV === 'development',
})

// Health check
export async function checkConnection() {
  const client = await pool.connect()
  try {
    await client.query('SELECT 1')
    return true
  } finally {
    client.release()
  }
}
```

## 4. CI/CD Patterns (GitHub Actions)

### Consensus Pipeline

```yaml
# .github/workflows/ci.yml
name: CI

on:
  pull_request:
    types: [opened, reopened, synchronize]
  push:
    branches: [main]

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}-${{ github.ref_protected && github.sha || '' }}
  cancel-in-progress: true

env:
  TURBO_TOKEN: ${{ secrets.TURBO_TOKEN }}
  TURBO_TEAM: ${{ secrets.TURBO_TEAM }}
  DO_NOT_TRACK: 1
  NEXT_TELEMETRY_DISABLED: 1

jobs:
  changes:
    uses: dorny/paths-filter@v3
    outputs:
      needs_build: ${{ steps.filter.outputs.needs_build }}
      needs_tests: ${{ steps.filter.outputs.needs_tests }}
    steps:
      - uses: actions/checkout@v4
      - id: filter
        uses: dorny/paths-filter@v3
        with:
          filters: |
            needs_build:
              - 'apps/**'
              - 'packages/**'
              - 'pnpm-lock.yaml'
              - 'package.json'
              - 'turbo.json'
              - 'tsconfig*.json'
            needs_tests:
              - 'apps/**'
              - 'packages/**'
              - 'pnpm-lock.yaml'
              - 'package.json'

  lint-typecheck:
    needs: changes
    if: needs.changes.outputs.needs_build == 'true'
    runs-on: ubuntu-24.04
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 9
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile
      - run: pnpm lint
      - run: pnpm typecheck

  test:
    needs: changes
    if: needs.changes.outputs.needs_tests == 'true'
    runs-on: ubuntu-24.04
    services:
      postgres:
        image: postgres:16
        env:
          POSTGRES_USER: test
          POSTGRES_PASSWORD: test
          POSTGRES_DB: test
        ports: 5432:5432
        options: >-
          --health-cmd="pg_isready -U test"
          --health-interval=10s
          --health-timeout=5s
          --health-retries=5
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 9
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile
      - run: pnpm db:generate
      - run: pnpm db:migrate
        env:
          DATABASE_URL: postgresql://test:test@localhost:5432/test
      - run: pnpm test

  build:
    needs: [lint-typecheck, test]
    if: needs.changes.outputs.needs_build == 'true'
    runs-on: ubuntu-24.04
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 9
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile
      - run: pnpm build
```
